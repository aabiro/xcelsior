# HTTPS — Reverse proxy to Fern docs
server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name docs.xcelsior.ca;

    # SSL certificates (Let's Encrypt)
    ssl_certificate /etc/letsencrypt/live/docs.xcelsior.ca/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/docs.xcelsior.ca/privkey.pem;

    # SSL configuration (Mozilla Modern)
    ssl_session_timeout 1d;
    ssl_session_cache shared:DocsSSL:10m;
    ssl_session_tickets off;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;

    # HSTS
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

    # Security headers
    add_header X-Frame-Options DENY always;
    add_header X-Content-Type-Options nosniff always;
    add_header Referrer-Policy strict-origin-when-cross-origin always;

    # Gzip
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml application/json application/javascript application/xml+rss text/javascript image/svg+xml text/x-component;

    # Rewrite tab-prefixed paths to flat paths (Fern tabs are client-side only)
    # /guides/quickstart -> /quickstart, /sdk/typescript-sdk -> /typescript-sdk
    rewrite ^/guides/(.+) /$1 break;
    rewrite ^/sdk/(.+) /$1 break;
    # Keep /api-reference as-is (Fern handles it as a redirect to auto-gen API docs)

    # Reverse proxy to Fern-hosted docs
    location / {
        proxy_pass https://xcelsior.docs.buildwithfern.com;
        proxy_ssl_server_name on;
        proxy_set_header Host xcelsior.docs.buildwithfern.com;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header Accept-Encoding "";

        # Pass through query params (RSC requests use ?_rsc=...)
        proxy_pass_request_headers on;

        # Timeouts
        proxy_connect_timeout 10s;
        proxy_read_timeout 30s;
        proxy_send_timeout 30s;

        # Buffering
        proxy_buffering on;
        proxy_buffer_size 16k;
        proxy_buffers 4 32k;

        # Handle redirects from upstream
        proxy_redirect https://xcelsior.docs.buildwithfern.com https://docs.xcelsior.ca;
    }
}

# configuration file /etc/nginx/sites-enabled/headscale:
# Headscale HTTPS Nginx Configuration
# Deploy to: /etc/nginx/sites-available/headscale
# Enable with: ln -s /etc/nginx/sites-available/headscale /etc/nginx/sites-enabled/
#
# NOT installed by scripts/deploy.sh. `hs.xcelsior.ca` resolves to a different
# host than the API, so a vhost for it on the API box answers no requests — and
# while this file still named a Let's Encrypt path, installing it there made
# `nginx -t` fail and aborted the whole deploy (2026-08-08).
#
# The certificate is a **self-signed internal cert**, ten years, not ACME. That
# is deliberate rather than a shortcut: the front reaches this service with
# `proxy_ssl_verify off`, so nothing validates the certificate and a publicly
# trusted one would buy nothing that this does not. The Let's Encrypt entry it
# replaced could not renew — the ACME challenge for this name is answered by the
# host DNS points at, not by the one that held the certificate — and its daily
# renewal failure was noise that masked the four *real* renewal failures found
# on 2026-08-08.
#
# The trade is worth stating plainly: TLS here is transport encryption, not
# identity. Anything that ever needs to *authenticate* this endpoint has to stop
# using `proxy_ssl_verify off` first, at which point this certificate must be
# pinned or replaced — it cannot chain to a public root.

server {
    listen 443 ssl http2;
    server_name hs.xcelsior.ca;

    ssl_certificate /etc/nginx/ssl/hs-internal.crt;
    ssl_certificate_key /etc/nginx/ssl/hs-internal.key;

    location / {
        proxy_pass http://127.0.0.1:8443;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_buffering off;
        proxy_read_timeout 600s;
    }
}
